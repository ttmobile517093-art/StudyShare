async function createRequest(requestType) { if (!currentUser) { alert("กรุณาเข้าสู่ระบบก่อนครับ"); window.location.href = "login.html"; return; } if (!currentPost) { alert("ไม่พบข้อมูลโพสต์ครับ"); return; } if (currentUser.uid === currentPost.ownerId) { alert("เจ้าของโพสต์ไม่สามารถขอหนังสือของตัวเองได้ครับ"); return; } if (currentPost.status === "reserved") { alert("รายการนี้ถูกจองแล้วครับ"); return; } let typeText = ""; if (requestType === "buy") { typeText = "🛒 ขอซื้อ"; } else if (requestType === "free") { typeText = "🎁 ขอรับ"; } else { typeText = "🔄 ขอแลก"; } const message = prompt( typeText + "\n\nฝากข้อความถึงเจ้าของหนังสือได้เลยครับ" ); if (message === null) { return; } if (!message.trim()) { alert("กรุณาใส่ข้อความก่อนส่งคำขอครับ"); return; } try { const userRef = doc( db, "users", currentUser.uid ); const userSnap = await getDoc(userRef); let userData = {}; if (userSnap.exists()) { userData = userSnap.data(); } const requestId = currentPost.id + "_" + currentUser.uid; const requestRef = doc( db, "requests", requestId ); const requestSnap = await getDoc(requestRef); if (requestSnap.exists()) { const oldRequest = requestSnap.data(); if (oldRequest.status === "pending") { alert( "คุณส่งคำขอรายการนี้ไปแล้ว และกำลังรอเจ้าของตอบครับ" ); return; } } await setDoc( requestRef, { postId: currentPost.id, postTitle: currentPost.title || "", postType: currentPost.type || "book", ownerId: currentPost.ownerId || "", ownerName: currentPost.ownerName || "", requesterId: currentUser.uid, requesterName: userData.name || currentUser.displayName || "นักเรียน", requesterStudentId: userData.studentId || "", requesterClass: userData.className || "", type: requestType, typeText: typeText, message: message.trim(), status: "pending", createdAt: serverTimestamp() } ); alert("ส่งคำขอเรียบร้อยแล้ว 🎉"); } catch (error) { console.error("ส่งคำขอไม่สำเร็จ:", error); alert( "ส่งคำขอไม่สำเร็จครับ\n" + error.message ); } }
