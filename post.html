async function createRequest(requestType) { try { // ตรวจสอบการเข้าสู่ระบบ if (!currentUser) { alert("กรุณาเข้าสู่ระบบก่อนครับ"); window.location.href = "login.html"; return; } // ตรวจสอบข้อมูลโพสต์ if (!currentPost) { alert("กำลังโหลดข้อมูลหนังสือ กรุณาลองใหม่อีกครั้งครับ"); return; } // เจ้าของห้ามขอของตัวเอง if (currentUser.uid === currentPost.ownerId) { alert("เจ้าของโพสต์ไม่สามารถขอหนังสือของตัวเองได้ครับ"); return; } // ตรวจสอบสถานะ if (currentPost.status === "reserved") { alert("รายการนี้ถูกจองแล้วครับ"); return; } // แปลงประเภทคำขอ let typeText = ""; if (requestType === "buy") { typeText = "🛒 ขอซื้อ"; } else if (requestType === "free") { typeText = "🎁 ขอรับ"; } else { typeText = "🔄 ขอแลก"; } // รับข้อความจากผู้ขอ const message = prompt( `${typeText}\n\nฝากข้อความถึงเจ้าของหนังสือได้เลยครับ` ); // กดยกเลิก if (message === null) { return; } // ป้องกันข้อความว่าง if (!message.trim()) { alert("กรุณาใส่ข้อความก่อนส่งคำขอครับ"); return; } // โหลดข้อมูลผู้ใช้ const userRef = doc( db, "users", currentUser.uid ); const userSnap = await getDoc(userRef); let userData = {}; if (userSnap.exists()) { userData = userSnap.data(); } // ID คำขอ const requestId = `${currentPost.id}_${currentUser.uid}`; const requestRef = doc( db, "requests", requestId ); // ตรวจคำขอเดิม const requestSnap = await getDoc(requestRef); if (requestSnap.exists()) { const oldRequest = requestSnap.data(); if (oldRequest.status === "pending") { alert( "คุณส่งคำขอรายการนี้ไปแล้วครับ 💜" ); return; } } // บันทึกคำขอ await setDoc( requestRef, { postId: currentPost.id, postTitle: currentPost.title || "", postType: currentPost.type || "book", ownerId: currentPost.ownerId || "", ownerName: currentPost.ownerName || "", requesterId: currentUser.uid, requesterName: userData.name || currentUser.displayName || "นักเรียน", requesterStudentId: userData.studentId || "", requesterClass: userData.className || "", type: requestType, typeText: typeText, message: message.trim(), status: "pending", createdAt: serverTimestamp() } ); // สำเร็จ alert( "ส่งคำขอเรียบร้อยแล้ว 🎉💜" ); } catch (error) { console.error( "REQUEST ERROR:", error ); alert( "ส่งคำขอไม่สำเร็จครับ\n\n" + error.message ); } }
